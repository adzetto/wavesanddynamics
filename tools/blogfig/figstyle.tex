  fs plot/.style={
    axis line style={line width=\FigThin, draw=fsLine},
    every tick/.style={line width=\FigThin, draw=fsLine},
    major tick length=2.5pt, minor tick length=1.5pt, tick align=inside,
    tick label style={font=\small},
    label style={font=\normalsize},
    legend style={draw=fsLine, line width=\FigThin, fill=white, font=\small,

    scaled ticks=false, clip mode=individual,

    xticklabel style={yshift=-1.2pt}, yticklabel style={xshift=-1.2pt},
    every axis label/.append style={inner sep=2pt},
    every legend/.append style={inner sep=2pt},
